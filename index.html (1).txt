        // ---------- بارگیری لە local storage ----------
        let completed = new Set();
        const STORAGE_KEY = 'dayflow_completed';

        function loadState() {
            try {
                const raw = localStorage.getItem(STORAGE_KEY);
                if (raw) {
                    const arr = JSON.parse(raw);
                    if (Array.isArray(arr)) {
                        completed = new Set(arr);
                        return;
                    }
                }
            } catch (_) {}
            completed = new Set();
        }

        function saveState() {
            localStorage.setItem(STORAGE_KEY, JSON.stringify([...completed]));
        }

        // ---------- کاتی نوێژ (هەولێر) ----------
        function getPrayerTimes() {
            const now = new Date();
            const day = now.getDate();
            const lat = 36.1911;
            const lng = 44.0092;
            const tz = 3;

            function calcTime(angle) {
                const decl = 23.44 * Math.sin((360/365) * (284 + day) * Math.PI/180);
                const latRad = lat * Math.PI/180;
                const decRad = decl * Math.PI/180;
                const cosHour = (Math.sin(angle * Math.PI/180) - Math.sin(latRad) * Math.sin(decRad)) / (Math.cos(latRad) * Math.cos(decRad));
                let hourAngle = Math.acos(Math.min(1, Math.max(-1, cosHour))) * 180/Math.PI;
                const solarNoon = 12 + (lng - 15*tz) / 15;
                return solarNoon - hourAngle/15;
            }

            function formatTime(dec) {
                const h = Math.floor(dec);
                const m = Math.round((dec - h) * 60);
                return `${String(h).padStart(2,'0')}:${String(m).padStart(2,'0')}`;
            }

            const fajr = calcTime(-18);
            const dhuhr = 12 + (lng - 15*tz) / 15;
            const asr = calcTime(4.5);
            const maghrib = calcTime(-0.833);
            const isha = calcTime(-17.5);

            return {
                fajr: formatTime(fajr),
                dhuhr: formatTime(dhuhr + 0.1),
                asr: formatTime(asr + 0.15),
                maghrib: formatTime(maghrib + 0.1),
                isha: formatTime(isha + 0.2)
            };
        }

            document.getElementById('maghribTime').textContent = t.maghrib;
            document.getElementById('ishaTime').textContent = t.isha;
        }
